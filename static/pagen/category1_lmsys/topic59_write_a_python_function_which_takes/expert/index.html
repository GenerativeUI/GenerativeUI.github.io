Python Implementation

🐍 Python 
def is_prime_and_even(n: int) -> bool:
    """Returns True if number is both prime AND even."""
    if n < 2:
        return False
    if n % 2 != 0:
        return False
    return n == 2


def is_prime(n: int) -> bool:
    """Check if number is prime."""
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False

    for i in range(3, int(n ** 0.5) + 1, 2):
        if n % i == 0:
            return False
    return True


# Test
print(is_prime_and_even(2))   # True
print(is_prime_and_even(4))   # False
